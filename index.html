<script>
let allGames = [];
let filteredGames = [];
let currentPage = 1;
const itemsPerPage = 12; // Lower number = faster load, less image spam

async function fetchGameList() {
try {
// In a real env, this fetches your local json
const response = await fetch('./games.json');
if (!response.ok) throw new Error("Database offline");
allGames = await response.json();
allGames.sort((a, b) => a.name.localeCompare(b.name));
filteredGames = [...allGames];
renderGames();
} catch (error) {
console.error(error);
document.getElementById('game-list').innerHTML = `<p class="col-span-full text-center text-red-500">Failed to connect to database. Check console.</p>`;
}
}

function renderGames() {
const list = document.getElementById('game-list');
const pageInfo = document.getElementById('page-info');
const totalPages = Math.ceil(filteredGames.length / itemsPerPage) || 1;
// Fix bounds if search reduces count
if (currentPage > totalPages) currentPage = totalPages;
const start = (currentPage - 1) * itemsPerPage;
const end = start + itemsPerPage;
const pageItems = filteredGames.slice(start, end);

list.innerHTML = '';
if (pageItems.length === 0) {
list.innerHTML = `<p class="col-span-full text-center py-20 text-zinc-600">No results found for that query.</p>`;
}

pageItems.forEach(game => {
const card = document.createElement('div');
card.className = 'game-card rounded-2xl cursor-pointer group';
card.onclick = () => launchGame(game);
// image logic
let img = game.image;
if (!img.startsWith('http') && !img.startsWith('/images')) {
img = img.startsWith('/') ? `/images${img}` : `/images/${img}`;
}

card.innerHTML = `
<div class="img-container rounded-t-2xl relative">
<img src="${img}" class="w-full h-full object-cover opacity-80 group-hover:opacity-100 transition-opacity" loading="lazy" onerror="this.src='https://placehold.co/400x250/0f0f0f/444?text=No+Preview'">
</div>
<div class="p-4">
<h3 class="font-bold text-sm truncate text-zinc-100">${game.name}</h3>
<p class="text-[10px] text-zinc-500 uppercase mt-1 tracking-widest">${game.type || 'Web App'}</p>
</div>
`;
list.appendChild(card);
});

pageInfo.innerText = `Page ${currentPage} of ${totalPages}`;
document.getElementById('prev-btn').disabled = currentPage === 1;
document.getElementById('next-btn').disabled = currentPage === totalPages;
}

function changePage(dir) {
currentPage += dir;
window.scrollTo({ top: 0, behavior: 'smooth' });
renderGames();
}

function handleSearch(query) {
filteredGames = allGames.filter(g =>
g.name.toLowerCase().includes(query.toLowerCase())
);
currentPage = 1; // reset to first page on search
renderGames();
}

function launchGame(game) {
const container = document.getElementById('immersive-container');
const iframe = document.getElementById('game-iframe');
document.getElementById('immersive-game-title').textContent = game.name;
iframe.src = game.directory;
container.style.display = 'flex';
document.body.style.overflow = 'hidden';
}

function closeImmersive() {
const container = document.getElementById('immersive-container');
const iframe = document.getElementById('game-iframe');
iframe.src = 'about:blank';
container.style.display = 'none';
document.body.style.overflow = 'auto';
}

function reloadGame() {
const iframe = document.getElementById('game-iframe');
const currentSrc = iframe.src;
iframe.src = 'about:blank';
setTimeout(() => { iframe.src = currentSrc; }, 10);
}

function toggleFullscreen() {
const container = document.getElementById('immersive-container');
if (!document.fullscreenElement) {
container.requestFullscreen();
} else {
document.exitFullscreen();
}
}

window.onload = () => {
fetchGameList();
document.getElementById('game-search-input').addEventListener('input', (e) => {
handleSearch(e.target.value);
});
};
